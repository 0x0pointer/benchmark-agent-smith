#!/usr/bin/env python3
"""Configure OpenCode for agent-smith benchmarks (run on each droplet)."""
import json
import shutil
import subprocess
import sys
from pathlib import Path

REPO_DIR = Path("/root/agent-smith")
CONFIG_DIR = Path.home() / ".config/opencode"
CONFIG_PATH = CONFIG_DIR / "opencode.json"
COMMANDS_DIR = CONFIG_DIR / "commands"

CONFIG_DIR.mkdir(parents=True, exist_ok=True)
COMMANDS_DIR.mkdir(parents=True, exist_ok=True)

# Write MCP server config
try:
    data = json.loads(CONFIG_PATH.read_text()) if CONFIG_PATH.exists() else {}
except Exception:
    data = {}

mcp = data.setdefault("mcp", {})
mcp["pentest-agent"] = {
    "type": "local",
    "command": [
        "/root/.local/bin/poetry", "-C", str(REPO_DIR),
        "run", "python", "-m", "mcp_server",
    ],
    "enabled": True,
    "timeout": 660000,
}

instructions = data.setdefault("instructions", [])
claude_md = str(REPO_DIR / "CLAUDE.md")
if claude_md not in instructions:
    instructions.append(claude_md)

CONFIG_PATH.write_text(json.dumps(data, indent=2) + "\n")
print(f"  OpenCode config written: {CONFIG_PATH}")

# Install all slash commands from skills/
# Flat .md files (e.g. pentester.md) → copy directly
# Skill directories (e.g. web-exploit/SKILL.md) → copy as <name>.md
skills_dir = REPO_DIR / "skills"
installed = []
if skills_dir.exists():
    for item in sorted(skills_dir.iterdir()):
        if item.is_file() and item.suffix == ".md":
            dest = COMMANDS_DIR / item.name
            shutil.copy2(item, dest)
            installed.append(item.name)
        elif item.is_dir():
            skill_file = item / "SKILL.md"
            if skill_file.exists():
                dest = COMMANDS_DIR / f"{item.name}.md"
                shutil.copy2(skill_file, dest)
                installed.append(f"{item.name}.md")
    print(f"  Installed {len(installed)} slash command(s): {', '.join(installed)}")
else:
    print(f"  Warning: skills dir not found at {skills_dir}")

print("  OpenCode setup complete")
